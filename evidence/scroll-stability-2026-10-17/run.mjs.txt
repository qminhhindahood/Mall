import {chromium} from 'file:///D:/Midterm/.scratch/impeccable-browser/node_modules/playwright/index.mjs';
import {readFile,writeFile} from 'node:fs/promises';
const out=new URL('file:///D:/Midterm/.scratch/0013-scroll-stability/evidence/scroll-stability-2026-10-17/');
const script=await readFile(new URL('reproduce.js.txt',out),'utf8');
const browser=await chromium.launch({headless:true,executablePath:'C:/Program Files/Google/Chrome/Application/chrome.exe',args:['--enable-unsafe-swiftshader']});
const results={recordedAt:new Date().toISOString(),browser:browser.version(),runs:[]};
try {
 for(const [mall,port] of [['saigon-centre',4313],['vincom-dong-khoi',4314]])for(const locale of ['en','vi'])for(const width of [1280,390]) {
  const context=await browser.newContext({viewport:{width,height:width===390?844:800},reducedMotion:'no-preference'});
  const page=await context.newPage();
  try {
   await page.goto(`http://localhost:${port}/${locale}`);
   await page.evaluate(script);
   await page.waitForFunction(()=>window.__scrollRun.status!=='running',null,{timeout:90000});
   const result=await page.evaluate(()=>window.__scrollRun);
   await page.evaluate(()=>{
    window.__wheel={overlays:[],samples:[]};let prior=false;
    window.__wheelObserver=new MutationObserver(()=>{const loading=!!document.querySelector('.loading-screen');if(loading&&!prior)__wheel.overlays.push({at:performance.now(),y:scrollY,href:location.href});prior=loading;});
    __wheelObserver.observe(document.body,{childList:true,subtree:true});
   });
   for(let cycle=0;cycle<2;cycle++)for(const direction of [1,-1])for(let step=0;step<12;step++){
    await page.mouse.wheel(0,direction*400);await page.waitForTimeout(35);
   }
   result.wheel=await page.evaluate(()=>{__wheelObserver.disconnect();return{...__wheel,floor:new URL(location.href).searchParams.get('floor'),loading:!!document.querySelector('.loading-screen')};});
   if(result.wheel.overlays.length)result.failures.push('Native wheel reversal reopens preparation overlay');
   if(result.wheel.floor!=='L2')result.failures.push('Native wheel reversal loses floor');
   if(result.failures.length)result.status='failed';
   results.runs.push({mall,locale,width,...result});
   console.log(JSON.stringify({mall,locale,width,status:result.status,overlays:result.overlays.length,wheelOverlays:result.wheel.overlays.length,failures:result.failures}));
   await page.screenshot({path:new URL(`${mall}-${locale}-${width}-after.png`,out).pathname.replace(/^\/(\w:)/,'$1')});
  }finally{await context.close();}
 }
}finally{await browser.close();await writeFile(new URL(process.argv[2]??'standalone.json',out),JSON.stringify(results,null,2)+'\n');}
if(results.runs.some(r=>r.status!=='passed'))process.exitCode=1;
